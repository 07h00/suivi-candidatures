function importerExcel(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(evt) { try { const data = new Uint8Array(evt.target.result); const workbook = XLSX.read(data, {type: 'array'}); const firstSheet = workbook.Sheets[workbook.SheetNames[0]]; // Convertit la feuille en objets JS const jsonRows = XLSX.utils.sheet_to_json(firstSheet, {defval: ""}); if (jsonRows.length === 0) { alert('⚠️ Le fichier semble vide.'); return; } let ajouts = 0; jsonRows.forEach(row => { const keys = Object.keys(row); // Récupère intelligemment l'entreprise et la ville peu importe le nom des colonnes let ent = row['Entreprise'] || row['entreprise'] || row['Nom'] || row['AGENCE'] || row['Agence'] || row[keys[0]] || ''; let ville = row['Ville'] || row['ville'] || row['VILLE'] || row['Lieu'] || row[keys[1]] || ''; let st = row['Statut'] || row['statut'] || row['ETAT'] || row['Etat'] || 'En attente'; let postule = parseInt(row['Postulé'] || row['Postule'] || row['Envois']) || 1; let relance = parseInt(row['Relances'] || row['Relance']) || 0; ent = ent.toString().trim(); ville = ville.toString().trim(); // Évite d'ajouter si c'est vide ou si c'est la ligne d'en-tête if (ent && ent.toLowerCase() !== 'entreprise' && ent.toLowerCase() !== 'nom') { candidatures.push({ ent, ville, st: st.toString(), postule, relance }); ajouts++; } }); if (ajouts > 0) { alert(`✅ ${ajouts} candidatures importées avec succès !`); sync(); } else { alert('⚠️ Impossible de lire les colonnes du fichier.'); } } catch (err) { alert('Erreur de lecture du fichier : ' + err.message); } }; reader.readAsArrayBuffer(file); }
